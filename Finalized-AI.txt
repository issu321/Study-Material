1. BFS (Breadth First Search) – Short Version

g = {'A':['B','C'], 'B':['D','E'], 'C':['F'], 'D':[], 'E':['F'], 'F':[]}
v = []
q = ['A']
while q:
    n = q.pop(0)
    if n not in v:
        v.append(n)
        q += g[n]
print("BFS:", v)

2. DFS (Depth First Search) – Short Version

g = {'A':['B','C'],'B':['D','E'],'C':['F'],'D':[],'E':['F'],'F':[]}
v = []

def dfs(n):
 if n not in v:
  v.append(n)
  for x in g[n]:
   dfs(x)

dfs('A')
print(v)

3. Best First Search – Short & Clean

g = {'A':[('B',1),('C',3)], 'B':[('D',4),('E',2)], 'C':[('F',5)], 'D':[], 'E':[('F',1)], 'F':[]}
v = []
q = [('A',0)]
while q:
    q.sort(key=lambda x:x[1])
    n = q.pop(0)[0]
    if n not in v:
        v.append(n)
        if n == 'F': break
        for x in g[n]:
            q.append(x)
print("Best:", v)

5. Tower of Hanoi – Simple 3 Disk

def hanoi(n, A, B, C):
    if n == 1:
        print("Move disk 1", A, "to", B)
    else:
        hanoi(n-1, A, C, B)
        print("Move disk", n, "from", A, "to", C)
        hanoi(n-1, B, A, C)
hanoi(3, 'A', 'B', 'C')
